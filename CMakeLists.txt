cmake_minimum_required(VERSION 3.20)
project(obs_lufs_meter VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OBS_SOURCE_ROOT "" CACHE PATH "OBS source SDK root")
set(OBS_IMPORT_LIBRARY "" CACHE FILEPATH "OBS import library")
set(OBS_FRONTEND_LIBRARY "" CACHE FILEPATH "OBS frontend import library")
option(BUILD_OBS_PLUGIN "Build native OBS dock" ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
add_library(ebur128 STATIC third_party/libebur128/ebur128.c)
target_include_directories(ebur128 PUBLIC third_party/libebur128 PRIVATE third_party/libebur128/queue)
target_compile_definitions(ebur128 PRIVATE _USE_MATH_DEFINES)
if(NOT MSVC)
  target_link_libraries(ebur128 PUBLIC m)
endif()
add_library(lufs_core STATIC src/meter.cpp)
target_include_directories(lufs_core PUBLIC src)
target_link_libraries(lufs_core PUBLIC ebur128)
enable_testing()
add_executable(meter_contract tests/meter_contract.cpp)
target_link_libraries(meter_contract PRIVATE lufs_core)
add_test(NAME meter_contract COMMAND meter_contract)
if(BUILD_OBS_PLUGIN)
  if(NOT WIN32)
    message(FATAL_ERROR "The OBS plugin currently supports Windows only. Use BUILD_OBS_PLUGIN=OFF for core tests.")
  endif()
  find_package(Qt6 REQUIRED COMPONENTS Widgets)
  if(NOT EXISTS "${OBS_SOURCE_ROOT}/libobs/obs.h" OR NOT EXISTS "${OBS_IMPORT_LIBRARY}" OR NOT EXISTS "${OBS_FRONTEND_LIBRARY}")
    message(FATAL_ERROR "Supply OBS_SOURCE_ROOT, OBS_IMPORT_LIBRARY, OBS_FRONTEND_LIBRARY")
  endif()
  add_library(lufs_ui STATIC src/dock.cpp)
  target_link_libraries(lufs_ui PUBLIC Qt6::Widgets)
  target_include_directories(lufs_ui PUBLIC src)
  enable_language(RC)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/third_party/libebur128/queue/sys/queue.h" QUEUE_HEADER)
  string(FIND "${QUEUE_HEADER}" "*/" QUEUE_NOTICE_END)
  math(EXPR QUEUE_NOTICE_LENGTH "${QUEUE_NOTICE_END} + 2")
  string(SUBSTRING "${QUEUE_HEADER}" 0 ${QUEUE_NOTICE_LENGTH} QUEUE_NOTICE)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/queue-license.txt" "${QUEUE_NOTICE}\n")
  configure_file(cmake/licenses.rc.in licenses.rc @ONLY)
  add_library(obs-lufs-meter MODULE src/plugin.cpp "${CMAKE_CURRENT_BINARY_DIR}/licenses.rc")
  target_include_directories(obs-lufs-meter PRIVATE "${OBS_SOURCE_ROOT}/libobs" "${OBS_SOURCE_ROOT}/frontend/api" "${CMAKE_CURRENT_SOURCE_DIR}/sdk")
  target_link_libraries(obs-lufs-meter PRIVATE lufs_core lufs_ui "${OBS_IMPORT_LIBRARY}" "${OBS_FRONTEND_LIBRARY}")
  set_target_properties(obs-lufs-meter PROPERTIES PREFIX "")
  add_executable(lufs_preview tests/preview.cpp)
  target_link_libraries(lufs_preview PRIVATE lufs_ui)
  add_executable(ui_contract tests/ui_contract.cpp)
  target_link_libraries(ui_contract PRIVATE lufs_ui)
  install(TARGETS obs-lufs-meter RUNTIME DESTINATION obs-plugins/64bit LIBRARY DESTINATION obs-plugins/64bit)
endif()
